name: Change Guard

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]
    # 'labeled'/'unlabeled' re-run the guard when the owner adds or removes 'human-approved'
    types: [opened, synchronize, reopened, labeled, unlabeled]

jobs:
  guard:
    name: Automatic Change Guard
    runs-on: ubuntu-latest

    steps:
      - name: Check out repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 0
          lfs: true

      - name: Set up Python 3.11
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: "pip"

      - name: Install backend dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r backend/requirements.txt
          pip install -r backend/requirements-dev.txt

      - name: Set up Node.js 20
        uses: actions/setup-node@v4
        with:
          node-version: "20"
          cache: "npm"
          cache-dependency-path: frontend/package-lock.json

      - name: Install frontend dependencies
        run: |
          cd frontend
          npm ci

      - name: Run Change Guard (Full Suite)
        env:
          # PR: compare with the PR base. Push: compare with the commit before the push, so every
          # pushed commit is checked (HEAD~1 only checked the last one). New branch: HEAD~1.
          BASE: ${{ github.event.pull_request.base.sha || github.event.before }}
          # Protected-path changes are accepted only on PRs the owner labelled 'human-approved'
          # (AI agents have no access to the GitHub UI), and on pushes to main, which must come from
          # merged PRs: enable branch protection "Require a pull request before merging" on main.
          HUMAN_APPROVED: ${{ (github.event_name == 'pull_request' && contains(github.event.pull_request.labels.*.name, 'human-approved')) || github.event_name == 'push' }}
        run: |
          if [ -z "$BASE" ] || [ "$BASE" = "0000000000000000000000000000000000000000" ]; then BASE="HEAD~1"; fi
          FLAGS="--full --base $BASE"
          if [ "$HUMAN_APPROVED" = "true" ]; then FLAGS="$FLAGS --human-approved"; fi
          python scripts/guard_changes.py $FLAGS
